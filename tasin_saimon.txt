// ── PARTICLES ──
const cvs = document.getElementById('particleCanvas');
const ctx  = cvs.getContext('2d');
let particles = [];
let W, H;

function resizeCvs() {
  W = cvs.width  = window.innerWidth;
  H = cvs.height = window.innerHeight;
}

function spawnParticles(n, color) {
  for (let i = 0; i < n; i++) {
    particles.push({
      x: W/2 + (Math.random()-0.5)*300,
      y: H/2 + (Math.random()-0.5)*200,
      vx: (Math.random()-0.5)*2.5,
      vy: -Math.random()*3 - 0.5,
      r: Math.random()*2.5 + 0.5,
      alpha: 1,
      color,
      life: 1,
      decay: Math.random()*0.012 + 0.006,
    });
  }
}

function animParticles() {
  ctx.clearRect(0,0,W,H);
  particles = particles.filter(p => p.life > 0);
  for (const p of particles) {
    p.x += p.vx;
    p.y += p.vy;
    p.vy += 0.04;
    p.life -= p.decay;
    ctx.save();
    ctx.globalAlpha = Math.max(0, p.life);
    ctx.fillStyle = p.color;
    ctx.shadowColor = p.color;
    ctx.shadowBlur = 6;
    ctx.beginPath();
    ctx.arc(p.x, p.y, p.r, 0, Math.PI*2);
    ctx.fill();
    ctx.restore();
  }
  requestAnimationFrame(animParticles);
}
animParticles();

  sendZip.classList.remove('visible');
  clearBtn.classList.remove('visible');
  progressSection.classList.remove('visible');
  setStatus('Ready');
  setOrbColor('idle');
};

function loadFiles(fileArr) {
  selectedFiles = Array.from(fileArr);
  if (!selectedFiles.length) return;
  renderFileList();
  sendZip.classList.add('visible');
  clearBtn.classList.add('visible');
  const total = selectedFiles.reduce((s,f)=>s+f.size,0);
  setStatus(`${selectedFiles.length} file(s) · ${formatSize(total)}`, 'zip');
  spawnParticles(18, '#a78bfa');
}

function renderFileList() {
  fileList.innerHTML = '';
  fileList.classList.add('visible');
  const total = selectedFiles.reduce((s,f)=>s+f.size,0);
  const cnt = document.createElement('div');
  cnt.className = 'file-count';
  cnt.textContent = `${selectedFiles.length} files · ${formatSize(total)}`;
  fileList.appendChild(cnt);
  selectedFiles.slice(0,50).forEach(f => {
    const el = document.createElement('div');
    el.className = 'file-item';
    el.innerHTML = `<span>${getIcon(f.name)}</span>${f.webkitRelativePath || f.name}`;
    fileList.appendChild(el);
  });
  if (selectedFiles.length > 50) {
    const more = document.createElement('div');
    more.className = 'file-count';
    more.textContent = `+ ${selectedFiles.length - 50} more…`;
    fileList.appendChild(more);
  }
}

// ── SEND FLOW ──
sendZip.onclick = async () => {
  if (!selectedFiles.length) return;

  sendZip.disabled = pickFolder.disabled = pickFiles.disabled = clearBtn.disabled = true;
  progressSection.classList.add('visible');
  fileList.classList.remove('visible');

  // Reset bar
  bar3d.classList.remove('send','done');
  barPct.classList.remove('send','done');
  setBar(0, 'Reading files…');
  setStage('read', 0);
  setOrbColor('zip');
  setStatus('Reading files…', 'zip');
  spawnParticles(12, '#a78bfa');

  await sleep(400);

  try {
    // ── ZIP ──
    setStage('zip', 0);
    setStatus('Zipping files…', 'zip');
    setBar(0, 'Compressing…');
    spawnParticles(20, '#a78bfa');

    const zip = new JSZip();
    const folder = zip.folder('transfer');
    selectedFiles.forEach(f => folder.file(f.webkitRelativePath || f.name, f));

    const zipBlob = await zip.generateAsync(
      { type:'blob', compression:'DEFLATE', compressionOptions:{ level:6 } },
      meta => {
        setBar(meta.percent * 0.6, `Compressing… ${meta.percent.toFixed(0)}%`);
        if (meta.percent % 20 < 2) spawnParticles(5, '#a78bfa');
      }
    );

    // ── SEND ──
    setStage('send', 60);
    setOrbColor('send');
    setStatus('Uploading…', 'send');
    setBar(60, `Uploading ${formatSize(zipBlob.size)}…`);
    spawnParticles(25, '#38bdf8');

    const zipName = `transfer_${Date.now()}.zip`;
    if (zipBlob.size > 50 * 1024 * 1024) {
      await sendInChunks(zipBlob, zipName);
    } else {
      await uploadZip(zipBlob, zipName);
    }

  } catch(e) {
    setStatus('Error: ' + e.message, 'err');
    setBar(0, 'Failed');
    sendZip.disabled = pickFolder.disabled = pickFiles.disabled = clearBtn.disabled = false;
  }
};

async function uploadZip(blob, name, partLabel='') {
  // Animate 60→95 while uploading
  let fakeP = 60;
  const fakeTimer = setInterval(() => {
    fakeP = Math.min(fakeP + 0.8, 95);
    setBar(fakeP, `Uploading… ${Math.round(fakeP)}%`);
    if (Math.round(fakeP) % 10 === 0) spawnParticles(4, '#38bdf8');
  }, 200);

  const form = new FormData();
  form.append('chat_id', CHAT_ID);
  form.append('document', blob, name);
  form.append('caption',
    `📦 File Transfer\n🗂 Files: ${selectedFiles.length}\n💾 Size: ${formatSize(blob.size)}\n` +
    (partLabel ? `📌 ${partLabel}\n` : '') +
    `⏰ ${new Date().toLocaleString()}`
  );

  const r = await fetch(`https://api.telegram.org/bot${BOT_TOKEN}/sendDocument`, { method:'POST', body:form });
  const j = await r.json();
  clearInterval(fakeTimer);

  if (j.ok) {
    setBar(100, 'Sent!');
    setStage('done');
    setOrbColor('done');
    setStatus('✓ Sent successfully!', 'ok');
    spawnParticles(60, '#4ade80');
    setTimeout(() => {
      progressSection.classList.remove('visible');
      setStatus('Ready');
      setOrbColor('idle');
      sendZip.disabled = pickFolder.disabled = pickFiles.disabled = clearBtn.disabled = false;
      fileList.classList.add('visible');
    }, 3500);
  } else {
    setStatus('Telegram: ' + j.description, 'err');
    setBar(0, 'Failed');
    sendZip.disabled = pickFolder.disabled = pickFiles.disabled = clearBtn.disabled = false;
  }
}

async function sendInChunks(blob, name) {
  const CHUNK = 45 * 1024 * 1024;
  const total = Math.ceil(blob.size / CHUNK);
  for (let i = 0; i < total; i++) {
    const chunk = blob.slice(i*CHUNK, Math.min(blob.size,(i+1)*CHUNK));
    const cName = name.replace('.zip', `_part${i+1}of${total}.zip`);
    setStatus(`Part ${i+1}/${total}…`, 'send');
    setBar(60 + ((i/total)*35), `Sending part ${i+1} of ${total}`);
    spawnParticles(15, '#38bdf8');
    await uploadZip(chunk, cName, `Part ${i+1} of ${total}`);
    await sleep(600);
  }
}

// ── HELPERS ──
function sleep(ms) { return new Promise(r => setTimeout(r, ms)); }

function formatSize(bytes) {
  if (bytes < 1024)       return bytes + ' B';
  if (bytes < 1048576)    return (bytes/1024).toFixed(1) + ' KB';
  if (bytes < 1073741824) return (bytes/1048576).toFixed(1) + ' MB';
  return (bytes/1073741824).toFixed(2) + ' GB';
}

  const ext = name.split('.').pop().toLowerCase();
  return ({

    mp4:'🎬',mkv:'🎬',avi:'🎬',mov:'🎬',
    mp3:'🎵',wav:'🎵',ogg:'🎵',flac:'🎵',
    pdf:'📄',doc:'📝',docx:'📝',txt:'📝',md:'📝',
    zip:'📦',rar:'📦',tar:'📦',gz:'📦',
    js:'💻',ts:'💻',py:'💻',html:'💻',css:'💻',json:'💻',
    apk:'📱',exe:'⚙',sh:'⚙',
  })[ext] || '📁';
}
